PC浏览器实现简易相机

1<template>
2  <Overlay :show="props.showCamera">
3    <div class="take-photo-container">
4      <video v-if="!photo" autoplay playsinline ref="videoRef"></video>
5      <img v-else :src="photo" style="object-fit: contain;width: 100%;height: 100%;" />
6    </div>
7    <div class="operate-group">
8      <Space>
9        <van-button @click="stop">结束拍照</van-button>
10        <van-button @click="reStartTakingPhoto" v-if="takePhotoStep == 2">重新拍照</van-button>
11        <van-button @click="useThis" v-if="takePhotoStep == 2">使用该照片</van-button>
12        <van-button @click="onConfirm" v-if="takePhotoStep == 1">拍照</van-button>
13      </Space>
14    </div>
15  </Overlay>
16</template>
17<script setup >
18import { Overlay, Space } from 'vant';
19import { watchEffect, watch } from 'vue';
20import { nextTick } from 'vue';
21import { ref, onMounted } from "vue"
22const props = defineProps(["showCamera"])
23const emits = defineEmits(["update:showCamera", "onUseThis"])
24const videoRef = ref()
25const photo = ref('');
26const takePhotoStep = ref(1)
27const mediaStreamRef = ref()
28function initMedia() {
29  return new Promise((s, e) => {
30    navigator?.mediaDevices?.getUserMedia({ audio: false, video: { facingMode: { exact: "environment" } } })
31    // navigator?.mediaDevices?.getUserMedia({ video: true })
32      .then((stream) => {
33        return s(stream)
34      })
35      .catch((err) => {
36        e(err)
37      });
38  })
39}
40async function initVideo() {
41  if (!videoRef.value) return
42  const mediaStream = await initMedia()
43  videoRef.value.srcObject = mediaStream
44  mediaStreamRef.value = mediaStream
45  videoRef.value.play()
46}
47
48function stop() {
49  stopVideo()
50  emits("update:showCamera", false)
51  takePhotoStep.value = 1
52  photo.value = ""
53}
54
55async function useThis() {
56  stopVideo()
57  emits("update:showCamera", false)
58  const blob = await (await fetch(photo.value)).blob()
59  const file = new File([blob], `${Date.now()}.png`, { type: blob.type })
60  emits("onUseThis", file)
61}
62
63function stopVideo() {
64  const mediaStream = mediaStreamRef.value
65  if (!mediaStream) return;
66  const tracks = mediaStream.getVideoTracks();
67  tracks.forEach(function (track) {
68    track.stop(); // 停止视频流的捕获
69  });
70  if (videoRef.value) {
71    videoRef.value.srcObject = null;
72  }
73}
74function reStartTakingPhoto() {
75  takePhotoStep.value = 1
76  photo.value = ''
77  nextTick(() => {
78    initVideo()
79  })
80}
81
82/**
83 * 利用canvas截图
84 */
85function onConfirm() {
86  takePhotoStep.value = 2
87  const canvas = document.createElement("canvas");
88  const video = videoRef.value;
89  const videoWith = video.videoWidth;
90  const videoHeight = video.videoHeight;
91  const minValue = Math.min(videoHeight, videoWith);
92
93
94  if (!video) return;
95  const ctx = canvas.getContext('2d');
96  canvas.width = 800;
97  canvas.height = 800;
98  let dxy = []
99  if (videoWith > videoHeight) {
100    dxy = [(videoWith - videoHeight) / 2, 0]
101  } else {
102    dxy = [0, (videoHeight - videoWith) / 2]
103  }
104
105  ctx.drawImage(video, ...dxy, minValue, minValue, 0, 0, 800, 800);
106  const imgUrl = canvas.toDataURL('image/png');
107  photo.value = imgUrl
108  stopVideo()
109}
110
111watchEffect(() => {
112  if (props.showCamera) {
113    reStartTakingPhoto()
114  }
115
116  // if (takePhotoStep.value != 1) {
117  //   stopVideo()
118  // }
119})
