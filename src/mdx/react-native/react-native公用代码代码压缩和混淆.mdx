# React Native公用代码压缩和混淆

减小第三方公用代码体积和防止代码重复率

## 方案 Rollup + Terser

### 依赖下载

```bash
    yarn add @babel/core @babel/preset-env @babel/runtime @react-native/babel-preset @rollup/plugin-babel @rollup/plugin-commonjs @rollup/plugin-json @rollup/plugin-node-resolve @rollup/plugin-terser @rollup/plugin-typescript rollup tslib typescript -D
```

### 相关配置文件
```js:rollup.config.js
    import resolve from "@rollup/plugin-node-resolve";
    import commonjs from "@rollup/plugin-commonjs";
    import babel from "@rollup/plugin-babel";
    import json from '@rollup/plugin-json'
    import terser from '@rollup/plugin-terser';
    import typescript from '@rollup/plugin-typescript';
    export default {
        input: ["./index.ts"],
        output: [
            {
                dir: "dist",
                format: 'cjs',
                plugins: [terser()]
            }
        ],
        plugins: [resolve(), commonjs(), babel(), json(), typescript()],
        external: ["react", "react-native"]
    };
```

```js:babel.config.js
module.exports = {
  presets: ["@babel/preset-env",],
};
```

```json:tsconfig.json
{
  "compilerOptions": {
    "jsx": "react",
    "target": "es2016", 
    "lib": [
      "es2021"
    ], 
    
    "module": "esnext", 
    "moduleResolution":"node",
    
    "declaration": true, 
    "outDir": "./dist", 
    "declarationDir": "./dist/@types", 
    "esModuleInterop": true,
    "forceConsistentCasingInFileNames": true, 
    "strict": true,
    "skipLibCheck": true 
  }
}
```
